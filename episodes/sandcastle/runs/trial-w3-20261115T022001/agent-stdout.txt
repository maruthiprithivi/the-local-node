```javascript
function slugify(input) {
  if (typeof input !== 'string') {
    input = String(input);
  }

  return input
    .toLowerCase()
    .trim()
    .replace(/[\s]+/g, '-')
    .replace(/[^a-z0-9-]/g, '')
    .replace(/-+/g, '-')
    .replace(/^-+|-+$/g, 'n-a');
}

module.exports = slugify;
```
