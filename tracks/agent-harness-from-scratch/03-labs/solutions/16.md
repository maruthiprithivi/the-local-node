```python
from course_harness.providers import OpenAIProvider, ProviderError
import pytest

def test_learner_no_implicit_live():
    provider = OpenAIProvider("fixture-model", api_key="dummy-secret")
    with pytest.raises(ProviderError) as error:
        provider.complete([{"role": "user", "content": "read only"}])
    assert error.value.category == "live_disabled"
    assert "dummy-secret" not in str(error.value)
```
