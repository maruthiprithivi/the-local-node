```python
from course_harness.advanced import MemoryStore

def test_learner_memory_scope(tmp_path):
    now = [100.0]
    memory = MemoryStore(tmp_path / "memory.db", lambda: now[0])
    try:
        memory.put("coding", "style", "use spaces", "toy README", 30)
        assert memory.retrieve("sre") == []
        assert memory.retrieve("coding")[0]["trust"] == "untrusted-memory"
        memory.delete("coding", "style")
        assert memory.retrieve("coding") == []
    finally:
        memory.close()
```
