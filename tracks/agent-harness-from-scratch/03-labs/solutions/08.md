from course_harness.tools import FixtureWorld, scenario_tools, Policy

def test_read_only_evidence():
    world = FixtureWorld(clock=lambda: 100)
    registry = scenario_tools(world)
    policy = Policy(allowed=frozenset({"observe_health"}))
    observed = world.observe({})
    assert observed["source"] == "synthetic_service"
    assert observed["timestamp"] == 100
    assert observed["fresh"] is True
    assert policy.decide(registry.tools["restart_service"],
                         {"service": "toy-service"}) == "deny"

## Exercise extension reference

```python
def label_observations(records, *, now, max_age=30):
    result = []
    for record in records:
        age = now - record["timestamp"]
        result.append({**record, "fresh": 0 <= age <= max_age,
                       "age_seconds": age, "trust": "untrusted"})
    return result

def test_stale_conflicting_observation():
    records = [
        {"source": "health", "timestamp": 100, "healthy": False},
        {"source": "metric", "timestamp": 1, "healthy": True},
    ]
    labeled = label_observations(records, now=100)
    assert labeled[0]["fresh"] is True
    assert labeled[1]["fresh"] is False
    assert labeled[1]["healthy"] is True  # Contradiction remains visible.
    assert all(record["trust"] == "untrusted" for record in labeled)

# A fake diagnosis should cite health at t=100 and mark metric at t=1 stale.
# This helper labels evidence; it neither proves a cause nor authorizes action.
```
