```python
from course_harness.team import Authenticator, TeamBoundary, TeamError
import pytest

def test_learner_expired_owner_cannot_complete(tmp_path):
    now = [100.0]
    permissions = frozenset({"task:submit", "task:work", "observe_health"})
    auth = Authenticator({"alice-token": ("alice", permissions)})
    alice = auth.authenticate("alice-token")
    boundary = TeamBoundary(tmp_path / "peers.db", auth, lambda: now[0],
                            memberships=[("alice", "course-team")])
    try:
        task = boundary.submit_task(alice, "course-team", "evidence-1",
                                    {"goal": "observe synthetic health"},
                                    {"observe_health"})
        lease = boundary.claim_task(alice, task,
                    worker_permissions={"observe_health"}, lease_seconds=1)
        now[0] = 102
        assert boundary.recover_tasks() == [task]
        with pytest.raises(TeamError):
            boundary.complete_task(alice, task, lease["fence"], {"health": "ok"})
        assert boundary.inspect_task(alice, task)["status"] == "uncertain"
    finally:
        boundary.close()
```
