def test_learner_changed_version_reloads():

    def load():
        seen.append(1)
        return {"count": len(seen)}
    assert cache.read("coding", "toy.py", "v1", "c1", load)["count"] == 1
    assert cache.read("coding", "toy.py", "v1", "c1", load)["count"] == 1
