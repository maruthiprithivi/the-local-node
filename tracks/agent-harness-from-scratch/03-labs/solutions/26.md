```python
from course_harness.persistence import DurableStore, StateError
from course_harness.resident import ResidentQueue
import pytest

def test_learner_stop_needs_explicit_reset(tmp_path):
    store = DurableStore(tmp_path / "resident.db", clock=lambda: 100)
    queue = ResidentQueue(store)
    queue.stop()
    with pytest.raises(StateError):
        queue.resume()
    assert queue.health()["human_stop"] is True
    queue.resume(clear_human_stop=True)
    assert queue.health()["human_stop"] is False
    store.close()
```
