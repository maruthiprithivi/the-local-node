```python
from course_harness.advanced import StreamAssembler

def test_learner_disconnect_discards_calls():
    validated = []
    stream = StreamAssembler(validated.append)
    stream.feed({"sequence": 0, "kind": "tool", "id": "a",
                 "name": "read", "delta": '{"path":"toy.py"}'})
    result = stream.finish()
    assert result.interrupted and result.reason == "disconnected"
    assert result.calls == () and validated == []
```
