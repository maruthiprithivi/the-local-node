from course_harness.advanced import MemoryStore

def test_learner_reopen_memory(tmp_path):
    path = tmp_path / "state.db"
    first = MemoryStore(path, lambda: 10)
    first.put("automation", "preference", "ask before overwrite", "user fixture", 60)

    finally:
        second.close()
