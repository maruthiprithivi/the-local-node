```python
def admit_burst(queue, events):
    from course_harness.resident import QueueFull
    admitted, rejected = [], []
    for key, payload in events:
        try:
            admitted.append(queue.enqueue(key, payload))
        except QueueFull:
            rejected.append(key)
    return {"admitted": admitted, "rejected": rejected}
```

```python
def test_learner_cancelled_stream_never_exposes_action():

    assert result.calls == () and result.reason == "cancelled"
```
