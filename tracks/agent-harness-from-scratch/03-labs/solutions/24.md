```python
from threading import Event
from course_harness.advanced import WorkerTask, coordinate

def test_learner_cancel_stops_worker_observations():
    cancel = Event()
    cancel.set()
    seen = []
    result = coordinate([WorkerTask("a", "coding")],
        lambda context: context.read("coding", lambda: seen.append("read")),
        cancel=cancel)
    assert result[0].status == "cancelled"
    assert seen == []
```
