```python
from course_harness.advanced import PromotionStore, BoundaryError
import pytest

def test_learner_changed_artifact_invalidates_review(tmp_path):
    store = PromotionStore(tmp_path / "versions.db", "max_steps=8")
    gates = {name: True for name in store.GATES}
    candidate = "max_steps=4"
    store.approve(candidate, gates, reviewer="synthetic-human-fixture",
                  reviewed_digest=store.digest(candidate))
    with pytest.raises(BoundaryError):
        store.promote(candidate + "\n")
    assert store.current() == "max_steps=8"
    store.close()
```
