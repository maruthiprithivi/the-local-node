```python
from course_harness.team import Authenticator, AuthContext, TeamBoundary, TeamError
import pytest

def test_learner_forged_caller_rejected(tmp_path):
    permissions = frozenset({"memory:read", "memory:write"})
    auth = Authenticator({"fixture-token": ("alice", permissions)})
    boundary = TeamBoundary(tmp_path / "team.db", auth, lambda: 100)
    try:
        forged = AuthContext("alice", permissions)
        with pytest.raises(TeamError):
            boundary.read_memory(forged)
        alice = auth.authenticate("fixture-token")
        boundary.put_memory(alice, "private", "use spaces", "user fixture", 60)
        assert boundary.read_memory(alice)[0]["owner"] == "alice"
    finally:
        boundary.close()
```
