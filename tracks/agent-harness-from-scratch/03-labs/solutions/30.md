```python
from course_harness.extensions import FakeGitHub, MentionReceiver
from course_harness.persistence import DurableStore
from course_harness.tools import canonical
import pytest

def test_learner_unprivileged_mention(tmp_path):
    store = DurableStore(tmp_path / "mentions.db", clock=lambda: 100)
    client = FakeGitHub()
    client.comments[1]["actor"] = "bob"
    receiver = MentionReceiver(store, client, secret=b"synthetic-only")
    event = {"repository": client.repo, "comment_id": 1, "revision": 1,
             "base": client.head, "fork": False, "action": "created"}
    body = canonical(event).encode()
    try:
        with pytest.raises(PermissionError):
            receiver.receive(body, receiver.sign(body))
        assert receiver.queue.inspect() == []
        assert client.prs == {}
    finally:
        store.close()
```
